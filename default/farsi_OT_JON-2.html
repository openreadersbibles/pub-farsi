וַיְמַ֤ןا

 וַיְהִ֤יث

 יֹונָה֙ج
اسم معلوم مذکر مفردיֹונָהیونس

 הַדָּ֔גح
اسم معلوم مذکر مفردדָּגماهی
 שְׁלֹשָׁ֥ה יָמִ֖ים וּשְׁלֹשָׁ֥ה לֵילֹֽות׃
۲  וַיִּתְפַּלֵּ֣לا
حیتپَعِل ویقتل سوم مذکر مفردפלל
 יֹונָ֔הب
اسم معلوم مذکر مفردיֹונָהیونس
 אֶל־יְהוָ֖ה אֱלֹהָ֑יו מִמְּעֵ֖יپ
اسم حنتوی مذکر جمعמֵעִיםشکم
 הַדָּגָֽהت
اسم معلوم مؤنث مفردדָּגָהماهی

۳  וַיֹּ֗אמֶרا
قَل ویقتل سوم مذکر مفردאמר
 קָ֠רָאתִיب
قَل کامل اول مفردקרא
 מִצָּ֥רָה לִ֛י אֶל־יְהוָ֖ה וַֽיַּעֲנֵ֑נִיپ

 מִבֶּ֧טֶן שְׁאֹ֛ול שִׁוַּ֖עְתִּיت
پیعِل کامل اول مفردשׁועفریاد زدن
 שָׁמַ֥עְתָּث
قَل کامل دوم مذکر مفردשׁמע
 קֹולִֽי׃
۴  וַתַּשְׁלִיכֵ֤נִיا
حیفیل ویقتل دوم مذکر مفرد پسوند ضمیری اول مفردשׁלך
 מְצוּלָה֙ب
اسم معلوم مؤنث مفردמְצֹולָהژرفا
 בִּלְבַ֣ב יַמִּ֔ים וְנָהָ֖ר יְסֹבְבֵ֑נִיپ
پیعِل یقتل سوم مذکر مفرد پسوند ضمیری اول مفردסבב
 כָּל־מִשְׁבָּרֶ֥יךָت
اسم معلوم مذکر جمع پسوند ضمیری دوم مذکر مفردמִשְׁבָּרموج
 וְגַלֶּ֖יךָث
اسم معلوم مذکر جمع پسوند ضمیری دوم مذکر مفردגַּלخیزاب
 עָלַ֥י עָבָֽרוּج
قَل کامل سوم جمعעבר

۵  וַאֲנִ֣י אָמַ֔רְתִּיا
قَل کامل اول مفردאמר
 נִגְרַ֖שְׁתִּיب

 מִנֶּ֣גֶד עֵינֶ֑יךָ אַ֚ךְ אֹוסִ֣יףپ

 לְהַבִּ֔יטت

 אֶל־הֵיכַ֖ל קָדְשֶֽׁךָ׃
۶  אֲפָפ֤וּנִיا
قَل کامل سوم جمع پسوند ضمیری اول مفردאפףمیان گرفتن
 מַ֨יִם֙ עַד־נֶ֔פֶשׁ תְּהֹ֖וםب
اسم معلوم مؤنث مفردתְּהֹוםژرفا
 יְסֹבְבֵ֑נִיپ
پیعِل یقتل سوم مذکر مفرد پسوند ضمیری اول مفردסבב
 ס֖וּףت
اسم معلوم مذکر مفردסוּףخزه
 חָב֥וּשׁث
قَل وصفی معلوم مذکر مفردחבשׁپیچیدن
 לְרֹאשִֽׁי׃
۷  לְקִצְבֵ֤יا
اسم حنتوی مذکر جمعקֶצֶבبُن
 הָרִים֙ יָרַ֔דְתִּיب
قَل کامل اول مفردירד
 הָאָ֛רֶץ בְּרִחֶ֥יהָپ
اسم معلوم مذکر جمع پسوند ضمیری سوم مؤنث مفردבְּרִיחַبسته
 בַעֲדִ֖י לְעֹולָ֑ם וַתַּ֧עַלت
حیفیل ویقتل دوم مذکر مفردעלה
 מִשַּׁ֛חַתث
اسم معلوم مؤنث مفردשַׁחַתچال
 חַיַּ֖י יְהוָ֥ה אֱלֹהָֽי׃
۸  בְּהִתְעַטֵּ֤ףا
حیتپَعِل مصدر حنتویעטףضعیف شدن
 עָלַי֙ נַפְשִׁ֔י אֶת־יְהוָ֖ה זָכָ֑רְתִּיب
قَل کامل اول مفردזכר
 וַתָּבֹ֤ואپ
قَل ویقتل سوم مؤنث مفردבוא
 אֵלֶ֨יךָ֙ תְּפִלָּתִ֔י אֶל־הֵיכַ֖ל קָדְשֶֽׁךָ׃
۹  מְשַׁמְּרִ֖יםا
پیعِل وصفی مجهول مذکر جمعשׁמר
 הַבְלֵי־שָׁ֑וְא חַסְדָּ֖ם יַעֲזֹֽבוּب
